Averaging Subcarrier Chunks

def shrink_csi(ratio, csi):
	subcarrier_count = csi.shape[1]
	assert(subcarrier_count % ratio == 0)
	shrunk_subcarrier_count = subcarrier_count // ratio
	csi_shrunk = []
		
	# Compute mean over chunks of subcarriers of size "ratio"
	for subcarrier in range(shrunk_subcarrier_count):
		shrunk_chunk = tf.math.reduce_mean(csi[:, (ratio * subcarrier):(ratio * (subcarrier + 1)), :], axis = 1)
		csi_shrunk.append(shrunk_chunk)
		
	return tf.stack(csi_shrunk, axis = 1)
